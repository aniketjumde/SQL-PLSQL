3.calculate bonus (Used Variable)
DELIMITER //
CREATE FUNCTION calculate_bonus(p_salary DECIMAL(10,2)) 
RETURNS DECIMAL(10,2) 
DETERMINISTIC 
BEGIN 
	DECLARE v_bonus DECIMAL(10,2); 
	SET v_bonus = p_salary * 0.10; 
	RETURN v_bonus; 
END //
